export const readKitVersion = (kitRoot = KIT_ROOT) => {
  try {
    return JSON.parse(fs.readFileSync(path.join(kitRoot, 'package.json'), 'utf-8')).version ?? 'unknown';
  } catch {
    return 'unknown'; // an unreadable package.json is reported by doctor; callers treat 'unknown' as unpinned
  }
};

export const readPackageVersion = (packageJsonPath = PACKAGE_JSON_PATH) => {
  try {
    return JSON.parse(fs.readFileSync(packageJsonPath, 'utf-8')).version || 'unknown';
  } catch (err) {
    process.stderr.write(`[mcp] cannot read version from ${packageJsonPath}: ${err.message}\n`);
    return 'unknown';
  }

export const resolveProjectName = (cwd = process.cwd()) => {
  const pkgPath = path.resolve(cwd, "package.json");
  const hasPkg = fs.existsSync(pkgPath);
  if (hasPkg) {
    try {
      const parsed = JSON.parse(fs.readFileSync(pkgPath, "utf-8"));
      return parsed.name || path.basename(cwd);
    } catch {
      return path.basename(cwd) || "Project";
    }
  }

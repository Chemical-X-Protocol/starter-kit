export const stripAnsi = (text) => {
  const isInvalidText = !text || typeof text !== "string";
  if (isInvalidText) return "";
  return text
    .replace(/\x1b\[[0-9;?]*[a-zA-Z]/g, "")
    .replace(/\x1b\][^\x07\x1b]*(\x07|\x1b\\)/g, "");
};

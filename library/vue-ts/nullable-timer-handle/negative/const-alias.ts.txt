// provenance: constructed from the alias form named in the engine doc (Library, THE CONFLICT RESOLUTION).
export function useAliasedTimeout(fn: () => void, delayMs: number) {
  let timerId: ReturnType<typeof setTimeout> | undefined;

  const stop = () => {
    const id = timerId;
    const hasTimer = id !== null;
    if (hasTimer) clearTimeout(id);
  };

  const start = () => {
    stop();
    timerId = setTimeout(fn, delayMs);
  };

  return { start, stop };
}
